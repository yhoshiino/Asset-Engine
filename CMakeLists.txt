cmake_minimum_required(VERSION 3.16)
project(AssetEngine CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(assetlib STATIC src/archive.cpp)
target_include_directories(assetlib PUBLIC include)
if(MSVC)
  target_compile_options(assetlib PUBLIC /W4)
else()
  target_compile_options(assetlib PUBLIC -Wall -Wextra -Wpedantic)
endif()

add_executable(asset_demo demo/main.cpp)
target_link_libraries(asset_demo PRIVATE assetlib)
add_executable(asset_tests tests/tests.cpp)
target_link_libraries(asset_tests PRIVATE assetlib)
add_executable(asset_bench bench/bench.cpp)
target_link_libraries(asset_bench PRIVATE assetlib)

enable_testing()
add_test(NAME asset_tests COMMAND asset_tests)

# Moteur3DCPU scene adapter: needs <Moteur3DCPU>/header (vec.h, obj.h, texture.h, ecs/, components/, stb_image.h)
#   cmake -S . -B build -DMOTEUR3D_HEADERS=/path/to/Moteur3DCPU/header
set(MOTEUR3D_HEADERS "" CACHE PATH "Path to Moteur3DCPU/header (enables scene adapter tests and demo)")
if(MOTEUR3D_HEADERS)
  add_executable(asset_scene_tests tests/scene_tests.cpp)
  target_link_libraries(asset_scene_tests PRIVATE assetlib)
  target_include_directories(asset_scene_tests PRIVATE ${MOTEUR3D_HEADERS})
  add_executable(asset_scene_demo demo/scene_demo.cpp)
  target_link_libraries(asset_scene_demo PRIVATE assetlib)
  target_include_directories(asset_scene_demo PRIVATE ${MOTEUR3D_HEADERS})
  add_test(NAME asset_scene_tests COMMAND asset_scene_tests)
endif()
